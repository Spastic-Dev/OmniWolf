// -------------------------------------------------------------------------
// 2. THE SQUID (DRONE)
// Description: Fast, erratically moving melee enemy.
// -------------------------------------------------------------------------

actor HoverSquid 
{
    Health 40                  // Roughly equal to a Guard
    Radius 24
    Height 64
    Speed 18                   // Faster than average Wolf3D guard
    PainChance 180
    Monster
    +SOLID
    +SHOOTABLE
    +AMBUSH
    +FLOORCLIP
    SeeSound "squid/sight"     // Define these in SNDINFO
    DeathSound "squid/death"
    AttackSound "squid/attack"
    States
    {
    Spawn:
        SQID A 10 A_Look
        Loop
    See:
        SQID A 3 A_Chase
        SQID A 3 A_Chase
        SQID B 3 A_Chase
        SQID B 3 A_Chase
        Loop
    Melee:
        SQID C 6 A_FaceTarget
        SQID D 6 A_CustomMelee(5, "squid/attack") // Deals 5 damage
        SQID C 6
        Goto See
    Pain:
        SQID E 3
        SQID E 3 A_Pain
        Goto See
    Death:
        SQID F 5
        SQID G 5 A_Scream
        SQID H 5 A_NoBlocking
        SQID I 5
        SQID J -1
        Stop
    }
}
